    display: grid;
    gap: 16px;
    grid-template-columns: repeat(auto-fit, minmax(280px, 1fr));
    align-items: start;      /* KEY: prevents vertical stretching */
    justify-items: start;    /* KEY: prevents horizontal stretching */
    justify-content: center; /* Centers grid properly */
    padding: 0 env(safe-area-inset-horizontal, 16px); /* SAFE AREA RESPECT */
  }
