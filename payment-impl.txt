import { z } from 'zod';
import { idSchema, positiveIntSchema, trackingNumberSchema } from './primitives';
import { emailSchema } from './auth';
import { CART_LIMITS, cartQuantitySchema } from './cart';

/**
 * Shipping / order limits. They mirror the existing customer checkout form rules
 * so the server never rejects what the storefront already accepts.
 */
export const ORDER_LIMITS = {
  name: { min: 2, max: 100 },
  phone: { pattern: /^[+]?[\d\s\-()]{7,15}$/ },
  address: { min: 3, max: 200 },
  city: { min: 2, max: 100 },
  state: { min: 2, max: 100 },
  zipCode: { min: 3, max: 20, pattern: /^[a-zA-Z0-9\s-]+$/ },
  country: { min: 2, max: 100 },
  notes: { max: 2000 },
  productName: { max: 300 },
  productImage: { max: 2048 },
  items: { min: 1, max: 100 },
} as const;

export const ORDER_STATUSES = [
  'PENDING',
  'CONFIRMED',
  'PROCESSING',
  'SHIPPED',
  'DELIVERED',
  'CANCELLED',
  'REFUNDED',
] as const;

export const SHIPPING_METHODS = ['vip', 'express', 'pickup'] as const;
export const shippingMethodSchema = z.enum(SHIPPING_METHODS);

const lettersPattern = /^[\p{L}\s'-]+$/u;

export const shippingNameSchema = z
  .string()
  .trim()
  .min(ORDER_LIMITS.name.min)
  .max(ORDER_LIMITS.name.max)
  .regex(lettersPattern);

export const shippingPhoneSchema = z.string().trim().regex(ORDER_LIMITS.phone.pattern);

/** Email is optional at checkout: empty string or a valid address. */
export const optionalEmailSchema = z.union([z.literal(''), emailSchema]);

export const shippingAddressSchema = z.object({
  customerName: shippingNameSchema,
  customerEmail: optionalEmailSchema,
  customerPhone: shippingPhoneSchema,
  shippingAddress: z
    .string()
    .trim()
    .min(ORDER_LIMITS.address.min)
    .max(ORDER_LIMITS.address.max),
  city: z
    .string()
    .trim()
    .min(ORDER_LIMITS.city.min)
    .max(ORDER_LIMITS.city.max)
    .regex(lettersPattern),
  state: z
    .string()
    .trim()
    .min(ORDER_LIMITS.state.min)
    .max(ORDER_LIMITS.state.max)
    .regex(lettersPattern),
  zipCode: z
    .string()
    .trim()
    .min(ORDER_LIMITS.zipCode.min)
    .max(ORDER_LIMITS.zipCode.max)
    .regex(ORDER_LIMITS.zipCode.pattern),
  country: z
    .string()
    .trim()
    .min(ORDER_LIMITS.country.min)
    .max(ORDER_LIMITS.country.max),
});

const moneySchema = z.number().finite().min(0);

export const orderItemSchema = z.object({
  productId: idSchema,
  sellerProductId: idSchema.nullish(),
  productName: z.string().trim().min(1).max(ORDER_LIMITS.productName.max),
  productImage: z.string().max(ORDER_LIMITS.productImage.max).nullish(),
  quantity: cartQuantitySchema,
  unitPrice: moneySchema,
  totalPrice: moneySchema,
  color: z.string().trim().max(CART_LIMITS.color.max).nullish(),
  size: z.string().trim().max(CART_LIMITS.size.max).nullish(),
});

export const createOrderSchema = shippingAddressSchema.extend({
  sellerId: positiveIntSchema,
  items: z
    .array(orderItemSchema)
    .min(ORDER_LIMITS.items.min)

  notes: z.string().max(ORDER_LIMITS.notes.max).nullish(),

import {

  OrderItem,
  OrderStatus,
  User,
  toSellerOrder,
} from '@kickzone-workspace/entities';
import {
  CreateOrderInput,
  OrdersConnection,
  SellerOrdersConnection,
  UpdateOrderInput,
  UpdateOrderStatusInput,
} from '@kickzone-workspace/inputs';
import {
  createOrderSchema,
  cursorSchema,
  idSchema,
  optionalIdSchema,
  paginationFirstSchema,
  trackingNumberSchema,
  updateOrderSchema,
  updateOrderStatusSchema,
} from '@kickzone-workspace/validation';
import { ZodValidationPipe } from './zod-validation.pipe';

const SELLER_STATUS_TRANSITIONS: Partial<Record<OrderStatus, OrderStatus>> = {
  [OrderStatus.CONFIRMED]: OrderStatus.PROCESSING,
  [OrderStatus.PROCESSING]: OrderStatus.SHIPPED,
  [OrderStatus.SHIPPED]: OrderStatus.DELIVERED,
};
import { OrderService, SellerService } from '@kickzone-workspace/services';
import { GraphqlAuthGuard } from '@kickzone-workspace/guards';
import {
  CurrentUser,
  type AuthenticatedUser,
  type OptionalAuthenticatedUser,
} from '@kickzone-workspace/decorators';

@Resolver(() => Order)
export class OrderResolver {
  constructor(
    private readonly orderService: OrderService,
    private readonly sellerService: SellerService
  ) {}

  /**
   * Extract user ID from various possible fields in the user object.
   */
  private getUserId(user: OptionalAuthenticatedUser): string {
    return String(user?.id ?? user?.sub ?? '');
  }

  /**
   * Check if the customer user owns the order by comparing emails.
   * Returns false if either email is empty/undefined to prevent empty-email matches.
   */
  private isCustomerOrderOwner(
    user: OptionalAuthenticatedUser,
    order: Order
  ): boolean {
    const userEmail = String(user?.email ?? '').trim().toLowerCase();
    const orderEmail = String(order?.customerEmail ?? '').trim().toLowerCase();
    return userEmail.length > 0 && orderEmail.length > 0 && userEmail === orderEmail;
  }

  /**
   * Normalize user role to uppercase standard values.
   */
  private getUserRole(user: OptionalAuthenticatedUser): 'ADMIN' | 'SELLER' | 'CUSTOMER' | 'UNKNOWN' {
    const rawRole = user?.role ?? user?.userType;
    if (!rawRole) return 'UNKNOWN';
    const normalized = String(rawRole).toUpperCase();
    if (normalized === 'ADMIN') return 'ADMIN';
    if (normalized === 'SELLER') return 'SELLER';
    if (normalized === 'CUSTOMER') return 'CUSTOMER';
    return 'UNKNOWN';
  }

  /**
   * Assert that the caller is an admin.
   */
  private assertAdmin(user: OptionalAuthenticatedUser): void {
    if (this.getUserRole(user) !== 'ADMIN') {
      throw new ForbiddenException('Admin access required');
    }
  }

  @Query(() => OrdersConnection)
  @UseGuards(GraphqlAuthGuard)
  async orders(
    @CurrentUser() user: AuthenticatedUser,
    @Args(
      'first',
      { type: () => Number, defaultValue: 10 },
      new ZodValidationPipe(paginationFirstSchema)
    )
    first: number,
    @Args(
      'after',
      { type: () => String, nullable: true },
      new ZodValidationPipe(cursorSchema)
    )
    after?: string,
    @Args(
      'sellerId',
      { type: () => String, nullable: true },
      new ZodValidationPipe(optionalIdSchema)
    )
    sellerId?: string,
    @Args('status', { type: () => OrderStatus, nullable: true })
    status?: OrderStatus
  ) {
    this.assertAdmin(user);
    return this.orderService.findAll(first, after, sellerId, status);
  }

  @Query(() => Order)
  @UseGuards(GraphqlAuthGuard)
  async order(
    @CurrentUser() user: AuthenticatedUser,
    @Args('id', { type: () => String }, new ZodValidationPipe(idSchema)) id: string
  ) {
    const order = await this.orderService.findOne(id);
    const role = this.getUserRole(user);

    switch (role) {
      case 'ADMIN':
        return order;

      case 'SELLER': {
        // Get seller info for this user to check ownership
        const seller = await this.sellerService.findOneByUserId(this.getUserId(user));
        if (!seller || order.sellerId !== seller.id) {
          throw new ForbiddenException('You do not have permission to view this order');
        }
        return toSellerOrder(order);
      }

      case 'CUSTOMER': {
        // Customer ownership is determined by email
        if (!this.isCustomerOrderOwner(user, order)) {
          throw new ForbiddenException('You do not have permission to view this order');
        }
        return order;
      }

      default:
        throw new ForbiddenException('Access denied');
    }
  }

  @Query(() => [Order])
  @UseGuards(GraphqlAuthGuard)
  async myOrders(
    @CurrentUser() user: User,
    @Args(
      'first',
      { type: () => Number, defaultValue: 10 },
      new ZodValidationPipe(paginationFirstSchema)
    )
    first: number,
    @Args(
      'after',
      { type: () => String, nullable: true },
      new ZodValidationPipe(cursorSchema)
    )
    after?: string
  ) {
    // For CUSTOMER: Get orders by customer email
    if (this.getUserRole(user) === 'CUSTOMER') {
      if (!user?.email || String(user.email).trim() === '') {
        return [];
      }
      const result = await this.orderService.getOrdersByCustomerEmail(
        user.email,
        first,
        after
      );
      return result.edges.map((edge) => edge.node);
    }

    // For SELLER: Use mySellerOrders query instead (returns SellerOrder without customer PII)
    throw new NotFoundException(
      'Sellers should use mySellerOrders query instead'
    );
  }

  /**
   * Seller orders query - returns SellerOrder type WITHOUT customer PII
   * Sellers should NOT see customer name, email, phone, or address
   */
  @Query(() => SellerOrdersConnection)
  @UseGuards(GraphqlAuthGuard)
  async mySellerOrders(
    @CurrentUser() user: User,
    @Args(
      'first',
      { type: () => Number, defaultValue: 10 },
      new ZodValidationPipe(paginationFirstSchema)
    )
    first: number,
    @Args(
      'after',
      { type: () => String, nullable: true },
      new ZodValidationPipe(cursorSchema)
    )
    after?: string,
    @Args('status', { type: () => OrderStatus, nullable: true })
    status?: OrderStatus
  ) {
    // Only sellers can access this
    if (this.getUserRole(user) !== 'SELLER') {
      throw new NotFoundException('Only sellers can access seller orders');
    }

    // Get seller info from user
    const seller = await this.sellerService.findOneByUserId(this.getUserId(user));
    if (!seller) {
      throw new NotFoundException('Seller not found for this user');
    }

    // Get orders for this seller
    const result = await this.orderService.getOrdersBySeller(
      seller.id.toString(),
      first,
      after,
      status
    );

    // Convert to SellerOrder connection (removes customer PII)
    return {
      edges: result.edges.map((edge) => ({
        cursor: edge.cursor,
        node: toSellerOrder(edge.node),
      })),
      pageInfo: result.pageInfo,
      totalCount: result.totalCount,
    };
  }

  @Mutation(() => Order)
  @UseGuards(GraphqlAuthGuard)
  async updateOrder(
    @CurrentUser() user: AuthenticatedUser,
    @Args('id', { type: () => String }, new ZodValidationPipe(idSchema)) id: string,

    const input: UpdateOrderInput & { items?: unknown } = updateOrderInput;

  @Mutation(() => Order)
  @UseGuards(GraphqlAuthGuard)
  async updateOrderStatus(
    @CurrentUser() user: AuthenticatedUser,
    @Args('updateOrderStatusInput', new ZodValidationPipe(updateOrderStatusSchema))
    updateOrderStatusInput: UpdateOrderStatusInput
  ) {
    const { id, status } = updateOrderStatusInput;
    const order = await this.orderService.findOne(id);

    const role = this.getUserRole(user);

    switch (role) {
      case 'ADMIN':
        // Admin can update any order status
        break;

      case 'SELLER': {
        // Get seller info for this user to check ownership
        const seller = await this.sellerService.findOneByUserId(this.getUserId(user));
        if (!seller || order.sellerId !== seller.id) {
          throw new ForbiddenException('You can only update status for your own seller orders');
        }
        // Sellers can only move orders through fulfilment steps
        if (SELLER_STATUS_TRANSITIONS[order.status] !== status) {
          throw new BadRequestException(
            `Sellers can only move orders through fulfilment steps (CONFIRMED->PROCESSING->SHIPPED->DELIVERED); ${order.status} -> ${status} is not allowed`
          );
        }
        break;
      }

      case 'CUSTOMER':
      default:
        throw new ForbiddenException('Customers are not allowed to update order status');
    }

    const updated = await this.orderService.updateOrderStatus(id, status);

    // If caller is SELLER, return redacted SellerOrder
    if (role === 'SELLER') {
      return toSellerOrder(updated);
    }

    return updated;
  }

  @Mutation(() => Boolean)
  @UseGuards(GraphqlAuthGuard)
  async cancelPendingOrder(
    @CurrentUser() user: AuthenticatedUser,
    @Args('orderId', { type: () => String }, new ZodValidationPipe(idSchema)) orderId: string
  ) {
    const order = await this.orderService.findOne(orderId);

    const role = this.getUserRole(user);

    // Ownership check
    let isOwner = false;
    if (role === 'ADMIN') {
      isOwner = true;
    } else if (role === 'CUSTOMER') {
      // Customer ownership by email (prevents empty-email matches)
      isOwner = this.isCustomerOrderOwner(user, order);
    } else if (role === 'SELLER') {
      // Seller ownership via SellerInfo
      const seller = await this.sellerService.findOneByUserId(this.getUserId(user));
      isOwner = seller != null && order.sellerId === seller.id;
    }

    if (!isOwner) {
      throw new ForbiddenException('You do not have permission to cancel this order');
    }

    // Status check: only PENDING orders can be cancelled
    if (order.status !== OrderStatus.PENDING) {
      throw new BadRequestException('Only pending orders can be cancelled');
    }

    await this.orderService.cancelPendingOrder(orderId);
    return true;
  }

  @Mutation(() => Boolean)
  @UseGuards(GraphqlAuthGuard)
  async deleteOrder(
    @CurrentUser() user: AuthenticatedUser,
    @Args('id', { type: () => String }, new ZodValidationPipe(idSchema)) id: string
  ) {
    this.assertAdmin(user);

    const order = await this.orderService.findOne(id);

    // If order is in an active paid status, cancel it first to restore stock
    const activePaidStatuses = [
      OrderStatus.CONFIRMED,
      OrderStatus.PROCESSING,
      OrderStatus.SHIPPED,
    ];

    if (activePaidStatuses.includes(order.status) || order.stockReserved === true) {
      await this.orderService.updateOrderStatus(id, OrderStatus.CANCELLED);
    }

    await this.orderService.deleteOrder(id);
    return true;
  }

  @Mutation(() => Order)
  @UseGuards(GraphqlAuthGuard)
  async updateOrderTracking(
    @CurrentUser() user: AuthenticatedUser,
    @Args('id', { type: () => String }, new ZodValidationPipe(idSchema)) id: string,
    @Args(
      'trackingNumber',
      { type: () => String },
      new ZodValidationPipe(trackingNumberSchema)
    )
    trackingNumber: string
  ) {
    const trimmed = String(trackingNumber ?? '').trim();
    if (!trimmed || trimmed.length > 100) {
      throw new BadRequestException('Tracking number must be 1-100 characters');
    }

    const order = await this.orderService.findOne(id);
    const role = this.getUserRole(user);

    switch (role) {
      case 'ADMIN':
        break;

      case 'SELLER': {
        const seller = await this.sellerService.findOneByUserId(this.getUserId(user));
        if (!seller || order.sellerId !== seller.id) {
          throw new ForbiddenException('You can only update tracking for your own seller orders');
        }
        break;
      }

      case 'CUSTOMER':
      default:
        throw new ForbiddenException('Customers are not allowed to update tracking');
    }

    const allowedStatuses = [OrderStatus.CONFIRMED, OrderStatus.PROCESSING, OrderStatus.SHIPPED];
    if (!allowedStatuses.includes(order.status)) {
      throw new BadRequestException('Tracking can only be set on confirmed, processing or shipped orders');
    }

    const updated = await this.orderService.updateTrackingNumber(id, trimmed);
    return role === 'SELLER' ? toSellerOrder(updated) : updated;
  }

  @Mutation(() => [Order])
  async recalculateBatchOrdersShipping(
    @Args('orderIds', { type: () => [String] }) orderIds: string[],
    @CurrentUser() user?: OptionalAuthenticatedUser,
    @Args('guestToken', { type: () => String, nullable: true }) guestToken?: string
  ): Promise<Order[]> {
    const uniqueOrderIds = Array.from(
      new Set(
        (orderIds ?? []).filter(
          (id): id is string => typeof id === 'string' && id.trim().length > 0
        )
      )
    );
    if (uniqueOrderIds.length === 0) {
      throw new BadRequestException('At least one orderId is required');
    }

    const orders = await Promise.all(
      uniqueOrderIds.map((id) => this.orderService.findOne(id))
    );

    const firstEmail = String(orders[0].customerEmail ?? '').trim().toLowerCase();
    const extractToken = (notes?: string | null) => {
      const m = /\[GuestToken:\s*([^\]]+)\]/.exec(notes ?? '');
      return m ? m[1].trim() : null;
    };
    const firstToken = extractToken(orders[0].notes);

    for (const order of orders) {
      if (order.status !== OrderStatus.PENDING) {
        throw new BadRequestException(
          `Order ${order.id} is not in PENDING status`
        );
      }
      if (String(order.customerEmail ?? '').trim().toLowerCase() !== firstEmail) {
        throw new BadRequestException(
          'All orders in a batch must belong to the same customer'
        );
      }
      if (extractToken(order.notes) !== firstToken) {
        throw new BadRequestException(
          'All orders in a batch must share the same guest checkout token'
        );
      }
      if (firstToken) {
        const provided = guestToken?.trim();
        if (!provided || provided !== firstToken) {
          throw new ForbiddenException('Invalid guest token for batch shipping recalculation');
        }

        const userEmail = String(user?.email ?? '').trim().toLowerCase();
        if (!userEmail || !firstEmail || firstEmail !== userEmail) {
          throw new ForbiddenException('Not allowed to recalculate shipping for this order batch');
        }
      }
    }

    return this.orderService.recalculateBatchOrdersShipping(orders);
  }

  @Mutation(() => ZarinPalPaymentResponse)
  async requestZarinPalPayment(
    @Args('input') input: ZarinPalPaymentRequestInput
  ) {
    // Normalize the list of order IDs from both orderId (singular) and orderIds (array)
    const targetOrderIds = Array.from(
      new Set([
        ...(input.orderIds ?? []),
        ...(input.order_id ? [input.order_id] : []),
      ].filter((id): id is string => typeof id === 'string' && id.trim().length > 0))
    );

    if (targetOrderIds.length === 0) {
      throw new BadRequestException('At least one orderId is required');
    }

    // Validate callback_url early - before any database lookups
    let parsedCallbackUrl: URL;
    try {
      parsedCallbackUrl = new URL(input.callback_url);
    } catch {
      throw new BadRequestException('Invalid callback_url');
    }
    if (
      parsedCallbackUrl.protocol !== 'http:' &&
      parsedCallbackUrl.protocol !== 'https:'
    ) {
      throw new BadRequestException('Invalid callback_url protocol');
    }

    const isSingleOrder = targetOrderIds.length === 1 && !input.orderIds;

    // Load all orders in targetOrderIds
    const orders = await Promise.all(
      targetOrderIds.map((id) => this.orderService.findOne(id))
    );

    // Verify all orders exist
    const missingOrders = targetOrderIds.filter(
      (_, index) => !orders[index]
    );
    if (missingOrders.length > 0) {
      if (isSingleOrder) {
        throw new BadRequestException('Order not found');
      }
      throw new BadRequestException(
        `Order(s) not found: ${missingOrders.join(', ')}`
      );
    }

    // Verify all orders are PENDING
    const nonPendingOrders = orders.filter(
      (order) => order.status !== OrderStatus.PENDING
    );
    if (nonPendingOrders.length > 0) {
      if (isSingleOrder) {
        throw new BadRequestException(
          `Order ${targetOrderIds[0]} is not in PENDING status and cannot be paid`
        );
      }
      throw new BadRequestException(
        `Order(s) ${nonPendingOrders
          .map((o) => o.id)
          .join(', ')} are not in PENDING status and cannot be paid`
      );
    }

    // Verify all orders belong to the same checkout customer/guest (by email)
    // Only for multi-order batches
    if (!isSingleOrder) {
      const firstOrder = orders[0];
      const firstCustomerEmail = String(firstOrder.customerEmail ?? '')
        .toLowerCase();
      const firstGuestToken = this.extractGuestToken(firstOrder.notes);

      for (const order of orders) {
        const customerEmail = String(order.customerEmail ?? '').toLowerCase();
        if (customerEmail !== firstCustomerEmail) {
          throw new BadRequestException(
            'All orders in a batch must belong to the same customer'
          );
        }
        const orderGuestToken = this.extractGuestToken(order.notes);
        if (firstGuestToken !== orderGuestToken) {
          throw new BadRequestException(
            'All orders in a batch must share the same guest checkout token'
          );
        }
      }
    }

    // Recompute server-authoritative shipping across the entire batch before summing totals
    if (typeof this.orderService.recalculateBatchOrdersShipping === 'function') {
      await this.orderService.recalculateBatchOrdersShipping(orders);
    }

    // Sum the USD totals across all orders in the batch
    const batchTotalUsd = orders.reduce(
      (sum, o) => sum + Number(o.total),
      0
    );

    // For single order, also validate the individual order total
    if (isSingleOrder) {
      const orderTotalUsd = Number(orders[0].total);
      if (!Number.isFinite(orderTotalUsd) || orderTotalUsd <= 0) {
        throw new BadRequestException('Order total is invalid or missing');
      }
    }

    // Fetch current USD -> IRR exchange rate (IRR represents Toman for ZarinPal)
    const rateRecord = await this.currencyService.getCurrentExchangeRate(
      'USD',
      'IRR'
    );
    const fxRate = Number(rateRecord?.rate ?? rateRecord);

    // Fail-closed FX guard: reject if rate is missing, not finite, or <= 1
    if (!Number.isFinite(fxRate) || fxRate <= 1) {
      throw new BadRequestException(
        'Exchange rate is not configured or invalid. Cannot process ZarinPal payment.'
      );
    }

    // Compute server-authoritative amount in Tomans
    const expectedBatchAmount = Math.round(batchTotalUsd * fxRate);

    // Client tampering check: if client sends amount, it must match expectedAmount within rounding tolerance
    if (input.amount != null) {
      const clientAmount = Number(input.amount);
      const tolerance = isSingleOrder
        ? Math.max(1, Math.round(expectedBatchAmount * 0.001)) // 0.1% or at least 1 Toman for single order
        : Math.max(
            10,
            targetOrderIds.length * 2
          ); // At least 10 Tomans or 2 per order for batch
      if (Math.abs(clientAmount - expectedBatchAmount) > tolerance) {
        if (isSingleOrder) {
          throw new BadRequestException('Payment amount does not match order total');
        }
        throw new BadRequestException(
          'Payment amount does not match batch order total'
        );
      }
    }

    // Always use server-computed expectedAmount for ZarinPal request
    const response = await this.paymentService.requestPayment({
      amount: expectedBatchAmount,
      callback_url: input.callback_url,
      description: input.description,
      currency: (input.currency as 'IRR' | 'IRT') || 'IRT',
      metadata: {
        mobile: input.mobile,
        email: input.email,
        order_id: targetOrderIds[0], // Primary order for metadata
      },
    });

    // Bind authority and server-computed expected amount to EVERY order in the batch
    for (const order of orders) {
      await this.orderService.setPaymentIntent({
        orderId: order.id,
        authority: response.data.authority,
        amount: expectedBatchAmount,
        currency: input.currency || 'IRT',
      });
    }

    return {
      code: response.data.code,
      message: response.data.message,
      authority: response.data.authority,
      fee_type: response.data.fee_type,
      fee: response.data.fee,
    };
  }

  private async executeVerifyZarinPalPayment(
    input: ZarinPalVerificationRequestInput
  ): Promise<ZarinPalVerificationResponse> {
    const orders = this.orderService.findOrdersByPaymentAuthority
      ? await this.orderService.findOrdersByPaymentAuthority(input.authority)
      : input.order_id
        ? [await this.orderService.findOne(input.order_id)].filter(
            (o): o is NonNullable<typeof o> => Boolean(o)
          )
        : [];

    if (orders.length === 0) {
      throw new BadRequestException(
        'No orders found for this payment authority'
      );
    }

    // Verify all orders have the matching authority
    for (const order of orders) {
      if (order.paymentAuthority !== input.authority) {
        throw new BadRequestException(
          'Payment authority does not match the order payment intent'
        );
      }
    }

    // Reject before calling the gateway if any order bound to this authority is already CANCELLED or REFUNDED
    const cancelledOrRefunded = orders.filter(
      (o) =>
        o.status === OrderStatus.CANCELLED || o.status === OrderStatus.REFUNDED
    );
    if (cancelledOrRefunded.length > 0) {
      throw new BadRequestException(
        `Cannot verify payment for cancelled or refunded order(s): ${cancelledOrRefunded
          .map((o) => o.id)
          .join(', ')}`
      );
    }

    // Pre-verify: ensure every PENDING order has sufficient stock on all items
    // This prevents one sibling order from confirming if another is out of stock
    const pendingOrders = orders.filter((o) => o.status === OrderStatus.PENDING);
    const stockRequirements = new Map<string, number>(); // sellerProductId -> requiredQty
    for (const order of pendingOrders) {
      for (const item of order.items ?? []) {
        const sp = (item as OrderItemWithSellerProduct).sellerProduct;
        if (sp?.id != null && typeof sp.stockQuantity === 'number') {
          const current = stockRequirements.get(sp.id) ?? 0;
          stockRequirements.set(sp.id, current + (item.quantity ?? 0));
        }
      }
    }
    for (const [sellerProductId, requiredQty] of stockRequirements.entries()) {
      // Find the sellerProduct to check current stock
      const sampleItem = pendingOrders
        .flatMap((o): OrderItemWithSellerProduct[] => o.items ?? [])
        .find((i) => i.sellerProduct?.id === sellerProductId);
      const available = sampleItem?.sellerProduct?.stockQuantity ?? 0;
      if (available < requiredQty) {
        throw new BadRequestException(
          `Insufficient stock for product ${sellerProductId}: required ${requiredQty}, available ${available}`
        );
      }
    }

    // Use the payment amount from the first order (they all share the same batch amount)
    const firstOrder = orders[0];
    if (
      firstOrder.paymentAmount === null ||
      firstOrder.paymentAmount === undefined
    ) {
      throw new BadRequestException(
        'Payment amount not recorded for this order. Please initiate payment again.'
      );
    }

    const amountToVerify = firstOrder.paymentAmount;

    const response = await this.paymentService.verifyPayment({
      amount: amountToVerify,
      authority: input.authority,
    });

    // If payment verification was successful (code 100 or 101)
    if (response.data.code === 100 || response.data.code === 101) {
      // Handle code 101 idempotently: if every order for this authority is already in a paid status
      // and has stock reserved, return success without re-decrementing stock.
      const paidStatuses = new Set<OrderStatus>([
        OrderStatus.CONFIRMED,
        OrderStatus.PROCESSING,
        OrderStatus.SHIPPED,
        OrderStatus.DELIVERED,
      ]);
      const allOrdersConfirmedAndStockReserved = orders.every(
        (o) => paidStatuses.has(o.status) && o.stockReserved === true
      );

      if (response.data.code === 101 && allOrdersConfirmedAndStockReserved) {
        // Idempotent success: already verified, stock already reserved
        return {
          code: response.data.code,
          message: response.data.message,
          ref_id: response.data.ref_id,
          card_pan: response.data.card_pan,
          card_hash: response.data.card_hash,
          fee_type: response.data.fee_type,
          fee: response.data.fee,
          orderIds: orders.map((o) => o.id),
          orderId: orders[0]?.id,
          orderNumber: orders.map((o) => o.orderNumber).join(', '),
          success: true,
        };
      }

      // Confirm ALL PENDING orders in deterministic order (id ASC)
      const pendingOrders = orders
        .filter((o) => o.status === OrderStatus.PENDING)
        .sort((a, b) => a.id.localeCompare(b.id));

      const confirmedOrderIds: string[] = [];
      for (const order of pendingOrders) {
        try {
          // Persist authority/amount before confirming to prevent tampering/replay
          await this.orderService.setPaymentIntent({
            orderId: order.id,
            authority: input.authority,
            amount: amountToVerify,
            currency: order.paymentCurrency || 'IRT',
          });

          await this.orderService.confirmOrderPayment(order.id);
          confirmedOrderIds.push(order.id);
        } catch (confirmError) {
          console.error(
            `Failed to finalize order ${order.id} after successful payment (authority: ${input.authority}, refId: ${response.data.ref_id}):`,
            confirmError
          );
          // Rollback any orders confirmed earlier in this batch
          for (const confirmedId of confirmedOrderIds) {
            try {
              if (this.orderService.updateOrderStatus) {
                await this.orderService.updateOrderStatus(confirmedId, OrderStatus.CANCELLED);
              }
            } catch (rollbackError) {
              console.error(
                `Rollback failed for order ${confirmedId}:`,
                rollbackError
              );
            }
          }
          // CRITICAL FIX: Return success: false when confirmOrderPayment throws.
          // The payment was captured by gateway (refId exists) but order confirmation/stock reservation failed.
          // Frontend must NOT clear cart and must show error with refId for support/refund.
          const errorMessage = confirmError instanceof Error
            ? `Payment was captured by gateway (Ref: ${response.data.ref_id}), but order confirmation failed: ${confirmError.message}`
            : `Payment was captured by gateway (Ref: ${response.data.ref_id}), but stock reservation failed.`;
          return {
            code: response.data.code,
            message: errorMessage,
            ref_id: response.data.ref_id,
            card_pan: response.data.card_pan,
            card_hash: response.data.card_hash,
            fee_type: response.data.fee_type,
            fee: response.data.fee,
            orderIds: orders.map((o) => o.id),
            orderId: orders[0]?.id,
            orderNumber: orders.map((o) => o.orderNumber).join(', '),
            success: false,
          };
        }
      }
    }

    // If payment verification failed (non-100/101), cancel ALL PENDING orders
    if (response.data.code !== 100 && response.data.code !== 101) {
      const pendingOrders = orders.filter(
        (o) => o.status === OrderStatus.PENDING
      );
      for (const order of pendingOrders) {
        try {
          await this.orderService.cancelPendingOrder(order.id);
        } catch (error) {
          // Log error but don't fail - payment verification response is what matters
          console.error(
            `Failed to cancel order ${order.id} after payment failure:`,
            error
          );
        }
      }
    }

    return {
      code: response.data.code,
      message: response.data.message,
      ref_id: response.data.ref_id,
      card_pan: response.data.card_pan,
      card_hash: response.data.card_hash,
      fee_type: response.data.fee_type,
      fee: response.data.fee,
      orderIds: orders.map((o) => o.id),
      orderId: orders[0]?.id,
      orderNumber: orders.map((o) => o.orderNumber).join(', '),
      success: response.data.code === 100 || response.data.code === 101,
    };
  }

  /**
   * Cancel a pending order when the buyer aborts on ZarinPal (Status=NOK).
   * Per ZarinPal docs, verify must not be called when Status is NOK.
   *
   * Two authorization paths:
   * 1. With non-empty `authority`: any caller (guest or authenticated) may cancel
   *    all PENDING orders bound to that authority. This preserves the existing
   *    guest-compatible behavior where the authority token proves possession.
   * 2. Without `authority` (empty/undefined): caller MUST provide `orderId` and be
   *    authorized as the order owner. Authorization succeeds if:
   *    - `user?.email` is non-empty and equals `order.customerEmail` (authenticated customer), OR
   *    - `guestToken` (trimmed, length >= 16) is found in `order.notes` as the literal
   *      substring `[GuestToken: <token>]` (guest who placed the order).
   *
   * @param orderId - Order ID (required when authority is not provided)
   * @param authority - ZarinPal authority token; if non-empty, authorizes via authority match
   * @param user - Current authenticated user (optional, from @CurrentUser decorator)
   * @param guestToken - Guest token for unauthenticated callers (optional)
   * @returns true if cancellation succeeded (idempotent for already-cancelled orders)
   * @throws BadRequestException if orderId missing without authority, or order not PENDING/already paid
   * @throws ForbiddenException if caller is not the order owner (without authority)
   */
  @Mutation(() => Boolean)
  async cancelPendingZarinPalOrder(
    @Args('orderId', { type: () => String, nullable: true }) orderId?: string,
    @Args('authority', { type: () => String, nullable: true }) authority?: string,
    @CurrentUser() user?: OptionalAuthenticatedUser,
    @Args('guestToken', { type: () => String, nullable: true }) guestToken?: string
  ) {
    // Path 1: Non-empty authority provided -> keep EXACT existing logic
    if (authority && authority.trim() !== '') {
      // Look up orders matching the authority
      const ordersByAuthority = await this.orderService.findOrdersByPaymentAuthority(authority);

      // If orderId was also passed, verify it matches the authority
      let matchedOrders = ordersByAuthority;
      if (orderId) {
        const orderById = await this.orderService.findOne(orderId);
        if (orderById.paymentAuthority !== authority) {
          throw new BadRequestException('Order does not match the provided payment authority');
        }
        // Ensure the orderById is in the matched list (in case findOrdersByPaymentAuthority missed it)
        if (!matchedOrders.some((o) => o.id === orderId)) {
          matchedOrders = [orderById, ...matchedOrders];
        }
      }

      if (matchedOrders.length === 0) {
        throw new BadRequestException('No matching order found for payment authority');
      }

      // Cancel each matched order that is PENDING
      for (const order of matchedOrders) {
        if (order.status === OrderStatus.CANCELLED) {
          // Idempotent: already cancelled, treat as success
          continue;
        }
        if (order.status !== OrderStatus.PENDING) {
          throw new BadRequestException(
            `Cannot cancel order ${order.id} with status ${order.status}. Only PENDING orders can be cancelled.`
          );
        }
        await this.orderService.cancelPendingOrder(order.id);
      }

      return true;
    }

    // Path 2: No authority (or empty) -> require orderId and ownership authorization
    if (!orderId || orderId.trim() === '') {
      throw new BadRequestException('Order ID is required when payment authority is not provided');
    }

    let order;
    try {
      order = await this.orderService.findOne(orderId);
    } catch (error) {
      if (error instanceof NotFoundException) {
        throw new ForbiddenException('Not allowed to cancel this order');
      }
      throw error;
    }

    // Authorization check
    const userEmail = user?.email?.trim();
    const providedGuestToken = guestToken?.trim();

    // Validate guestToken format: no ']' or newlines to prevent injection
    if (providedGuestToken && /[\]\r\n]/.test(providedGuestToken)) {
      throw new BadRequestException('Invalid guest token format');
    }

    const isOwnerByEmail = Boolean(userEmail && userEmail.length > 0 && order.customerEmail?.toLowerCase() === userEmail.toLowerCase());

    const isOwnerByGuestToken = Boolean(
      providedGuestToken &&
      providedGuestToken.length >= 16 &&
      (order.notes ?? '').includes(`[GuestToken: ${providedGuestToken}]`)
    );

    if (!isOwnerByEmail && !isOwnerByGuestToken) {
      throw new ForbiddenException('Not allowed to cancel this order');
    }

    // Status checks
    if (order.status === OrderStatus.CANCELLED) {
      // Idempotent: already cancelled, treat as success
      return true;
    }

    if (order.status !== OrderStatus.PENDING || order.paymentVerifiedAt) {
      throw new BadRequestException(
        `Cannot cancel order ${order.id} with status ${order.status}. Only unpaid PENDING orders can be cancelled.`
      );
    }

    await this.orderService.cancelPendingOrder(order.id);
    return true;
  }

      where: { id: createOrderInput.sellerId },
    });

    if (!seller) {
      throw new NotFoundException(
        `Seller with ID ${createOrderInput.sellerId} not found`,
      );
    }

    order.sellerStoreName = seller.storeName;
    order.sellerPhone = seller.phone || undefined;
    order.sellerAddress = seller.address || undefined;

    // Validate and compute prices from catalog
    const validatedItems = await Promise.all(
      inputItems.map(async (item) => {
        // Validate quantity
        if (!Number.isInteger(Number(item.quantity)) || Number(item.quantity) <= 0) {
          throw new BadRequestException(
            `Quantity must be a positive integer for product ${item.productId}`,
          );
        }

        if (!item.sellerProductId) {
          throw new BadRequestException(
            `sellerProductId is required for product ${item.productId}`,
          );
        }

        const sellerProduct = await this.sellerProductRepository.findOne({
          where: { id: item.sellerProductId },
          relations: ['seller', 'productInfo'],
        });

        if (!sellerProduct) {
          throw new NotFoundException(
            `SellerProduct with ID ${item.sellerProductId} not found`,
          );
        }

        // Ownership check: sellerProduct must belong to the order's seller
        if (Number(sellerProduct.seller.id) !== createOrderInput.sellerId) {
          throw new BadRequestException(
            `Product ${item.sellerProductId} does not belong to seller ${createOrderInput.sellerId}`,
          );
        }

        // Stock check
        if (sellerProduct.stockQuantity < item.quantity) {
          throw new BadRequestException(
            `Insufficient stock for product ${item.sellerProductId}. Available: ${sellerProduct.stockQuantity}, Requested: ${item.quantity}`,
          );
        }

        // Compute catalog price: price * (1 - discount/100)
        const catalogUnitPrice = Number(
          (sellerProduct.price * (1 - (sellerProduct.discount || 0) / 100)).toFixed(2),
        );
        const catalogTotalPrice = Number((catalogUnitPrice * item.quantity).toFixed(2));

        return {
          ...item,
          sellerProduct,
          catalogUnitPrice,
          catalogTotalPrice,
          originalPrice: sellerProduct.price,
          discountPercentage: sellerProduct.discount,
          productInfoId: sellerProduct.productInfoId || undefined,
        };
      }),
    );

    // Aggregate stock check: sum quantities per sellerProductId across all line items
    const requestedBySellerProduct = new Map<
      string,
      { totalQty: number; available: number }
    >();
    for (const item of validatedItems) {
      const sellerProductId = item.sellerProductId!; // validated as required earlier
      const current = requestedBySellerProduct.get(sellerProductId) ?? {
        totalQty: 0,
        available: item.sellerProduct.stockQuantity,
      };
      current.totalQty += Number(item.quantity);
      if (current.totalQty > current.available) {
        throw new BadRequestException(
          `Insufficient stock for product ${sellerProductId}. Available: ${current.available}, Requested: ${current.totalQty}`,
        );
      }
      requestedBySellerProduct.set(sellerProductId, current);
    }

    // Compute server-side totals from catalog prices
    order.subtotal = Number(
      validatedItems.reduce((sum, item) => sum + item.catalogTotalPrice, 0).toFixed(2),
    );

    // Validate and normalize shipping method
    const shippingMethod = this.normalizeShippingMethod(
      createOrderInput.shippingMethod,
    );

    // Read seller's shipping payment mode
    const paymentMode =
      seller.shippingPaymentMode ?? ShippingPaymentMode.PAY_ON_DELIVERY;

    order.shippingMethod = shippingMethod;
    order.shippingPaymentMode = paymentMode;

    let shippingCost = 0;
    if (paymentMode === ShippingPaymentMode.PAY_ONLINE) {
      let baseFee = Number(seller.shippingFee || 0);
      if (this.shippingCalculationService) {
        const calc = await this.shippingCalculationService.calculateShippingCost(
          order.subtotal,
          [
            {
              sellerId: seller.id,
              shippingFee: baseFee,
              sellerName: seller.storeName,
              shippingPaymentMode: ShippingPaymentMode.PAY_ONLINE,
            },
          ],
        );
        baseFee = calc.totalShippingCost;
      }
      shippingCost = applyShippingMethodMultiplier(baseFee, shippingMethod);
    }

    order.shippingCost = shippingCost;
    order.total = Number((order.subtotal + order.shippingCost).toFixed(2));

    const savedOrder = await this.orderRepository.save(order);

    const orderItemsToCreate = validatedItems.map((item) => ({
      orderId: savedOrder.id,
      productId: item.productId,
      sellerProductId: item.sellerProductId,
      productName: item.productName,
      productImage: item.productImage,
      quantity: item.quantity,
      unitPrice: item.catalogUnitPrice,
      totalPrice: item.catalogTotalPrice,
      color: item.color,
      size: item.size,
      originalPrice: item.originalPrice,
      discountPercentage: item.discountPercentage,
      productInfoId: item.productInfoId,
    }));

    const savedItems = await this.orderItemRepository.save(orderItemsToCreate);

    savedOrder.items = savedItems;

    return savedOrder;
  }

  private normalizeShippingMethod(
    method?: string | null,
  ): 'vip' | 'express' | 'pickup' {
    if (method === undefined || method === null || method === '') {
      return 'vip';
    }
    const normalized = String(method).trim().toLowerCase();
    if (
      normalized === 'vip' ||
      normalized === 'express' ||
      normalized === 'pickup'
    ) {
      return normalized;
    }
    throw new BadRequestException(`Invalid shipping method: ${method}`);
  }

  async recalculateBatchOrdersShipping(orders: Order[]): Promise<Order[]> {
    if (!Array.isArray(orders) || orders.length === 0) {
      return [];
    }

    // Load SellerInfo for each order if not already attached
    const sellerMap = new Map<number, SellerInfo>();
    for (const order of orders) {
      if (order.seller) {
        sellerMap.set(Number(order.sellerId), order.seller);
      } else if (!sellerMap.has(Number(order.sellerId))) {
        const found = await this.sellerInfoRepository.findOne({
          where: { id: Number(order.sellerId) },
        });
        if (found) {
          sellerMap.set(Number(order.sellerId), found);
        }
      }
    }

    const onlineOrders: Order[] = [];
    const codOrders: Order[] = [];

    for (const order of orders) {
      const seller = sellerMap.get(Number(order.sellerId));
      const mode =
        seller?.shippingPaymentMode ??
        order.shippingPaymentMode ??
        ShippingPaymentMode.PAY_ON_DELIVERY;

      if (mode === ShippingPaymentMode.PAY_ONLINE) {
        onlineOrders.push(order);
      } else {
        codOrders.push(order);
      }
    }

    // 1. Force all PAY_ON_DELIVERY orders to 0 shipping
    for (const order of codOrders) {
      const subtotal = Number(Number(order.subtotal).toFixed(2));
      const changed =
        Number(order.shippingCost) !== 0 ||
        Number(order.total) !== subtotal ||
        order.shippingPaymentMode !== ShippingPaymentMode.PAY_ON_DELIVERY;

      order.shippingPaymentMode = ShippingPaymentMode.PAY_ON_DELIVERY;
      order.shippingCost = 0;
      order.total = subtotal;

      if (changed) {
        await this.orderRepository.save(order);
      }
    }

    // 2. Recompute online shipping across all PAY_ONLINE orders in deterministic sellerId ASC order
    if (onlineOrders.length > 0) {
      const sortedOnline = [...onlineOrders].sort((a, b) => {
        const diff = Number(a.sellerId) - Number(b.sellerId);
        return diff !== 0 ? diff : String(a.id).localeCompare(String(b.id));
      });

      const onlineSubtotal = sortedOnline.reduce(
        (sum, o) => sum + Number(o.subtotal),
        0,
      );

      const sellerFeesInput = sortedOnline.map((o) => {
        const seller = sellerMap.get(Number(o.sellerId));
        return {
          sellerId: Number(o.sellerId),
          shippingFee: Number(seller?.shippingFee ?? 0),
          sellerName: seller?.storeName ?? o.sellerStoreName,
          shippingPaymentMode: ShippingPaymentMode.PAY_ONLINE,
        };
      });

      const breakdownBySellerId = new Map<string, number>();
      if (this.shippingCalculationService) {
        const calc =
          await this.shippingCalculationService.calculateShippingCost(
            onlineSubtotal,
            sellerFeesInput,
          );
        for (const item of calc.shippingBreakdown) {
          breakdownBySellerId.set(String(item.sellerId), item.appliedFee);
        }
      } else {
        for (const item of sellerFeesInput) {
          breakdownBySellerId.set(String(item.sellerId), item.shippingFee);
        }
      }

      for (const order of sortedOnline) {
        const appliedBaseFee =
          breakdownBySellerId.get(String(order.sellerId)) ?? 0;
        const method = this.normalizeShippingMethod(order.shippingMethod);
        const computedShipping = applyShippingMethodMultiplier(
          appliedBaseFee,
          method,
        );
        const computedTotal = Number(
          (Number(order.subtotal) + computedShipping).toFixed(2),
        );

        const changed =
          Number(order.shippingCost) !== computedShipping ||
          Number(order.total) !== computedTotal ||
          order.shippingPaymentMode !== ShippingPaymentMode.PAY_ONLINE ||
          order.shippingMethod !== method;

        order.shippingMethod = method;
        order.shippingPaymentMode = ShippingPaymentMode.PAY_ONLINE;
        order.shippingCost = computedShipping;
        order.total = computedTotal;

        if (changed) {
          await this.orderRepository.save(order);
        }
      }
    }

    return orders;
  }

  async findAll(
    first = 10,
    after?: string,
    sellerId?: string,
    status?: OrderStatus,
  ): Promise<OrdersConnection> {
    const queryBuilder = this.orderRepository
      .createQueryBuilder('order')
      .leftJoinAndSelect('order.items', 'items')
      .leftJoinAndSelect('order.seller', 'seller')
      .orderBy('order.createdAt', 'DESC')
      .addOrderBy('order.id', 'DESC');

    if (sellerId) {
      const sellerIdInt = parseInt(sellerId, 10);
      if (isNaN(sellerIdInt)) {
        throw new BadRequestException('Invalid sellerId');
      }
      queryBuilder.andWhere('order.sellerId = :sellerId', {
        sellerId: sellerIdInt,
      });
    }

    if (status) {
      queryBuilder.andWhere('order.status = :status', { status });
    }

    if (after !== undefined && after !== null) {
      const cursor = await this.decodeCursor(after);
      queryBuilder.andWhere('order.createdAt < :cursor', { cursor });
    }

    const orders = await queryBuilder.take(first + 1).getMany();

    const hasNextPage = orders.length > first;
    const edges: OrderEdge[] = orders.slice(0, first).map((order) => ({
      node: order,
      cursor: this.encodeCursor(order.createdAt),
    }));

    return {
      edges,
      pageInfo: {
        hasNextPage,
        hasPreviousPage: after !== undefined && after !== null && after !== '',
        endCursor: edges[edges.length - 1]?.cursor,
      },
      totalCount: await this.getTotalCount(sellerId, status),
    };
  }

      order.customerPhone = updateOrderInput.customerPhone;
    }

      order.customerName = updateOrderInput.customerName;
    }
    if (updateOrderInput.shippingAddress !== undefined) {

    }

      order.country = updateOrderInput.country;
    }
    if (updateOrderInput.notes !== undefined) {
      order.notes = updateOrderInput.notes;
    }
    if (updateOrderInput.trackingNumber !== undefined) {
      order.trackingNumber = updateOrderInput.trackingNumber;
    }

    return this.orderRepository.save(order);
  }

  async updateOrderStatus(id: string, status: OrderStatus): Promise<Order> {
    return this.runInTransaction(async (manager) => {
      const orderRepo = manager
        ? manager.getRepository(Order)
        : this.orderRepository;
      const sellerProductRepo = manager
        ? manager.getRepository(SellerProduct)
        : this.sellerProductRepository;

      const order = await orderRepo.findOne({
        where: { id },
        relations: ['items'],
        lock: manager ? { mode: 'pessimistic_write' } : undefined,
      });

      if (!order) {
        throw new NotFoundException(`Order with ID ${id} not found`);
      }

      if (order.status === status) {
        return order;
      }

      const allowedTransitions =
        OrderService.ALLOWED_STATUS_TRANSITIONS[order.status] || [];
      if (!allowedTransitions.includes(status)) {
        throw new BadRequestException(
          `Invalid status transition from ${order.status} to ${status}`,
        );
      }

      if (
        order.status === OrderStatus.PENDING &&
        status === OrderStatus.CONFIRMED &&
        !order.stockReserved
      ) {
        await this.decreaseStock(order, sellerProductRepo);
        order.stockReserved = true;
      }

      if (
        (status === OrderStatus.CANCELLED || status === OrderStatus.REFUNDED) &&
        order.stockReserved
      ) {
        await this.restoreStock(order, sellerProductRepo);
        order.stockReserved = false;
      }

      order.status = status;
      return orderRepo.save(order);
    });
  }

  async deleteOrder(id: string): Promise<void> {
    return this.runInTransaction(async (manager) => {
      const orderRepo = manager
        ? manager.getRepository(Order)
        : this.orderRepository;
      const sellerProductRepo = manager
        ? manager.getRepository(SellerProduct)
        : this.sellerProductRepository;

      const order = await orderRepo.findOne({
        where: { id },
        relations: ['items'],
        lock: manager ? { mode: 'pessimistic_write' } : undefined,
      });

      if (!order) {
        throw new NotFoundException(`Order with ID ${id} not found`);
      }

      if (order.stockReserved) {
        await this.restoreStock(order, sellerProductRepo);
      }

      await orderRepo.remove(order);
    });
  }

  private generateOrderNumber(): string {
    const timestamp = Date.now().toString();
    const random = Math.random().toString(36).substring(2, 8).toUpperCase();
    return `ORD-${timestamp}-${random}`;
  }

  private generateTrackingNumber(): string {
    const now = new Date();
    const year = now.getFullYear();
    const month = String(now.getMonth() + 1).padStart(2, '0');
    const day = String(now.getDate()).padStart(2, '0');
    const random = Math.random().toString(36).substring(2, 8).toUpperCase();
    return `KZ-${year}${month}${day}-${random}`;
  }

  private encodeCursor(date: Date): string {
    return Buffer.from(date.toISOString()).toString('base64');
  }

  private async decodeCursor(cursor: string): Promise<Date> {
    if (typeof cursor !== 'string' || cursor.trim() === '') {
      throw new BadRequestException('Invalid cursor');
    }
    try {
      const decoded = Buffer.from(cursor, 'base64').toString();
      const date = new Date(decoded);
      if (!decoded || Number.isNaN(date.getTime())) {
        throw new BadRequestException('Invalid cursor');
      }
      return date;
    } catch {
      throw new BadRequestException('Invalid cursor');
    }
  }

  private async decreaseStock(
    order: Order,
    sellerProductRepo: Repository<SellerProduct>,
  ): Promise<void> {
    if (order.stockReserved) {
      return;
    }

    for (const item of order.items) {
      if (
        !Number.isInteger(Number(item.quantity)) ||
        Number(item.quantity) <= 0
      ) {
        throw new BadRequestException(
          'Order item quantity must be a positive integer',
        );
      }
    }

    const qtyBySellerProduct = new Map<string, number>();
    for (const item of order.items) {
      const sellerProductId = item.sellerProductId || item.productId;
      const existing = qtyBySellerProduct.get(sellerProductId) || 0;
      qtyBySellerProduct.set(sellerProductId, existing + Number(item.quantity));
    }

    const sortedSellerProductIds = Array.from(qtyBySellerProduct.keys()).sort();

    for (const sellerProductId of sortedSellerProductIds) {
      const requiredQty = qtyBySellerProduct.get(sellerProductId)!;

      const sellerProduct = await sellerProductRepo.findOne({
        where: { id: sellerProductId },
        lock: { mode: 'pessimistic_write' },
      });

      if (!sellerProduct) {
        throw new NotFoundException(
          `SellerProduct with ID ${sellerProductId} not found`,
        );
      }

      if (sellerProduct.stockQuantity < requiredQty) {
        throw new BadRequestException(
          `Insufficient stock for product ${sellerProductId}. Available: ${sellerProduct.stockQuantity}, Requested: ${requiredQty}`,
        );
      }
    }

    for (const sellerProductId of sortedSellerProductIds) {
      const requiredQty = qtyBySellerProduct.get(sellerProductId)!;

      const sellerProduct = await sellerProductRepo.findOne({
        where: { id: sellerProductId },
        lock: { mode: 'pessimistic_write' },
      });

      if (sellerProduct) {
        sellerProduct.stockQuantity -= requiredQty;
        await sellerProductRepo.save(sellerProduct);
      }
    }
  }

  async confirmOrderPayment(
    orderId: string,
    refId?: string,
  ): Promise<Order> {
    return this.runInTransaction(async (manager) => {
      const orderRepo = manager
        ? manager.getRepository(Order)
        : this.orderRepository;
      const sellerProductRepo = manager
        ? manager.getRepository(SellerProduct)
        : this.sellerProductRepository;

      const order = await orderRepo.findOne({
        where: { id: orderId },
        relations: ['items'],
        lock: manager ? { mode: 'pessimistic_write' } : undefined,
      });

      if (!order) {
        throw new NotFoundException(`Order with ID ${orderId} not found`);
      }

      if (order.status === OrderStatus.CONFIRMED && order.stockReserved) {
        return order;
      }

      if (order.status !== OrderStatus.PENDING) {
        throw new BadRequestException(
          `Cannot confirm payment for order in status ${order.status}`,
        );
      }

      await this.decreaseStock(order, sellerProductRepo);
      order.stockReserved = true;
      order.status = OrderStatus.CONFIRMED;
      order.paymentVerifiedAt = new Date();

      return orderRepo.save(order);
    });
  }

  async findOrdersByPaymentAuthority(authority: string): Promise<Order[]> {
    if (!authority) return [];
    const orders = await this.orderRepository.find({
      where: { paymentAuthority: authority },
      relations: ['items', 'seller'],
      order: { id: 'ASC' },
    });

    // Collect unique non-empty sellerProductId from all order items
    const sellerProductIds = new Set<string>();
    for (const order of orders) {
      for (const item of order.items || []) {
        if (item.sellerProductId) {
          sellerProductIds.add(item.sellerProductId);
        }
      }
    }

    if (sellerProductIds.size > 0) {
      const sellerProducts = await this.sellerProductRepository.findBy({
        id: In(Array.from(sellerProductIds)),
      });
      const sellerProductMap = new Map(sellerProducts.map((sp) => [sp.id, sp]));

      for (const order of orders) {
        for (const item of order.items || []) {
          (item as OrderItem & { sellerProduct?: SellerProduct | null }).sellerProduct =
            item.sellerProductId ? sellerProductMap.get(item.sellerProductId) ?? null : null;
        }
      }
    }

    return orders;
  }

  async cancelPendingOrder(orderId: string): Promise<Order> {
    return this.runInTransaction(async (manager) => {
      const orderRepo = manager
        ? manager.getRepository(Order)
        : this.orderRepository;
      const sellerProductRepo = manager
        ? manager.getRepository(SellerProduct)
        : this.sellerProductRepository;

      const order = await orderRepo.findOne({

      order.status = OrderStatus.CANCELLED;
      return orderRepo.save(order);
    });

  private async restoreStock(
    order: Order,
    sellerProductRepo: Repository<SellerProduct>,
  ): Promise<void> {
    if (!order.stockReserved) {
      return;
    }

    for (const item of order.items) {
      if (!Number.isInteger(Number(item.quantity)) || Number(item.quantity) <= 0) {
        throw new BadRequestException(
          'Order item quantity must be a positive integer',
        );
      }
    }

    const qtyBySellerProduct = new Map<string, number>();
    for (const item of order.items) {
      const sellerProductId = item.sellerProductId || item.productId;
      const existing = qtyBySellerProduct.get(sellerProductId) || 0;
      qtyBySellerProduct.set(sellerProductId, existing + Number(item.quantity));
    }

    const sortedSellerProductIds = Array.from(qtyBySellerProduct.keys()).sort();

    for (const sellerProductId of sortedSellerProductIds) {
      const requiredQty = qtyBySellerProduct.get(sellerProductId)!;

      const sellerProduct = await sellerProductRepo.findOne({
        where: { id: sellerProductId },

                          '0 10px 24px -4px rgba(0, 0, 0, 0.28), inset 0 1px 1px 0 rgba(255, 255, 255, 0.35)',

                            'linear-gradient(135deg, #2c2c28 0%, #050505 100%)',

                  letterSpacing: '0.02em',
                  textTransform: 'none',

                  transition: 'all 0.3s cubic-bezier(0.16, 1, 0.3, 1)',

    </CustomersLayout>
