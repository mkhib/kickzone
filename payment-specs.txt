import {
  ORDER_STATUSES,
  createOrderSchema,
  shippingAddressSchema,
  updateOrderSchema,
  updateOrderStatusSchema,
} from './order';

const shipping = {
  customerName: 'Ali Rezaei',
  customerEmail: '',
  customerPhone: '+98 912 123 4567',
  shippingAddress: 'Valiasr St, No 12',
  city: 'Tehran',
  state: 'Tehran',
  zipCode: '1234567890',
  country: 'Iran',
};

const item = {
  productId: '101',
  sellerProductId: '301',
  productName: 'Air Jordan 1',
  productImage: null,
  quantity: 2,
  unitPrice: 100,
  totalPrice: 200,
  color: 'Black',
  size: '42',
};

const order = { ...shipping, sellerId: 1, items: [item], notes: '', shippingCost: 0 };

describe('shippingAddressSchema', () => {
  it('accepts a valid address with an empty optional email', () => {
    expect(shippingAddressSchema.safeParse(shipping).success).toBe(true);
  });

  it('accepts Farsi letters in name, city and state', () => {
    const fa = { ...shipping, customerName: 'علی رضایی', city: 'تهران', state: 'تهران' };
    expect(shippingAddressSchema.safeParse(fa).success).toBe(true);
  });

  it('accepts a valid email and keeps its case', () => {
    const r = shippingAddressSchema.parse({ ...shipping, customerEmail: ' Ali@Example.com ' });
    expect(r.customerEmail).toBe('Ali@Example.com');
  });

  it.each([
    ['customerName', 'A'],
    ['customerName', 'Ali123'],
    ['customerEmail', 'not-an-email'],
    ['customerPhone', '12'],
    ['customerPhone', 'abc-defg-hij'],
    ['shippingAddress', 'ab'],
    ['city', 'T1'],
    ['state', ''],
    ['zipCode', '12'],
    ['zipCode', '12#45'],
    ['country', 'I'],
  ])('rejects invalid %s=%j', (field, value) => {
    expect(shippingAddressSchema.safeParse({ ...shipping, [field]: value }).success).toBe(false);
  });

  it('trims surrounding whitespace', () => {
    const r = shippingAddressSchema.parse({ ...shipping, city: '  Tehran  ' });
    expect(r.city).toBe('Tehran');
  });
});

describe('createOrderSchema', () => {
  it('accepts a valid order, including a guest-token note', () => {
    const r = createOrderSchema.safeParse({ ...order, notes: '[GuestToken: abc]' });
    expect(r.success).toBe(true);
  });

  it('rejects an order without items', () => {
    expect(createOrderSchema.safeParse({ ...order, items: [] }).success).toBe(false);
  });

  it('rejects sellerId 0, negative shipping cost and negative prices', () => {
    expect(createOrderSchema.safeParse({ ...order, sellerId: 0 }).success).toBe(false);
    expect(createOrderSchema.safeParse({ ...order, shippingCost: -1 }).success).toBe(false);
    expect(
      createOrderSchema.safeParse({ ...order, items: [{ ...item, unitPrice: -5 }] }).success
    ).toBe(false);
  });

  it('rejects item quantity 0 and above the cart cap', () => {
    expect(createOrderSchema.safeParse({ ...order, items: [{ ...item, quantity: 0 }] }).success).toBe(false);
    expect(createOrderSchema.safeParse({ ...order, items: [{ ...item, quantity: 1000 }] }).success).toBe(false);
  });

  it('allows a missing shipping cost and notes', () => {
    const { notes: _n, shippingCost: _s, ...rest } = order;
    expect(createOrderSchema.safeParse(rest).success).toBe(true);
  });
});

describe('updateOrderSchema', () => {
  it('accepts an empty patch and all-null fields', () => {
    expect(updateOrderSchema.safeParse({}).success).toBe(true);
    expect(
      updateOrderSchema.safeParse({ city: null, notes: null, trackingNumber: null, items: null }).success
    ).toBe(true);
  });

  it('keeps items so the resolver can still reject item rewrites', () => {
    const r = updateOrderSchema.parse({ items: [item] });
    expect(r.items).toHaveLength(1);
  });

  it('rejects an invalid field in the patch', () => {
    expect(updateOrderSchema.safeParse({ zipCode: '1' }).success).toBe(false);
    expect(updateOrderSchema.safeParse({ trackingNumber: '' }).success).toBe(false);
  });
});

describe('updateOrderStatusSchema', () => {
  it('accepts every known status', () => {
    for (const status of ORDER_STATUSES) {
      expect(updateOrderStatusSchema.safeParse({ id: 'o-1', status }).success).toBe(true);
    }
  });

  it('rejects unknown status and empty id', () => {
    expect(updateOrderStatusSchema.safeParse({ id: 'o-1', status: 'LOST' }).success).toBe(false);
    expect(updateOrderStatusSchema.safeParse({ id: ' ', status: 'PENDING' }).success).toBe(false);
  });
});

describe('PaymentResolver - Batch Shipping Recomputation & Amount Guard', () => {
  function setupResolver() {
    const ordersDb: Record<string, Partial<Order>> = {
      'ord-cod': {
        id: 'ord-cod',
        orderNumber: 'ORD-1',
        customerEmail: 'buyer@example.com',
        status: OrderStatus.PENDING,
        sellerId: 1,
        subtotal: 100,
        shippingCost: 15, // stale/poisoned value before recalculation
        total: 115,
        shippingMethod: 'vip',
        shippingPaymentMode: ShippingPaymentMode.PAY_ON_DELIVERY,
      },
      'ord-online': {
        id: 'ord-online',
        orderNumber: 'ORD-2',
        customerEmail: 'buyer@example.com',
        status: OrderStatus.PENDING,
        sellerId: 2,
        subtotal: 50,
        shippingCost: 0, // client tried to send 0
        total: 50,
        shippingMethod: 'express', // 45% of $10 = $4.50
        shippingPaymentMode: ShippingPaymentMode.PAY_ONLINE,
      },
    };

    const paymentService = {
      requestPayment: jest.fn(async () => ({
        data: {
          code: 100,
          message: 'Success',
          authority: 'A00000000000000000000000000000000000',
        },
        errors: [],
      })),
    };

    const orderService = {
      findOne: jest.fn(async (id: string) => ordersDb[id] ?? null),
      recalculateBatchOrdersShipping: jest.fn(async (orders: Partial<Order>[]) => {
        for (const o of orders) {
          if (o.id === 'ord-cod') {
            o.shippingCost = 0;
            o.total = 100;
          } else if (o.id === 'ord-online') {
            o.shippingCost = 4.5;
            o.total = 54.5;
          }
        }
        return orders as Order[];
      }),
      setPaymentIntent: jest.fn(async () => ({})),
    };

    const currencyService = {
      getCurrentExchangeRate: jest.fn(async () => ({ rate: 60000 })),
    };

    const resolver = new PaymentResolver(
      paymentService as unknown as PaymentService,
      orderService as unknown as OrderService,
      currencyService as unknown as CurrencyService,
    );

    return { resolver, paymentService, orderService };
  }

  it('recomputes batch totals (COD = 0, Online Express = 4.50) and requests exact server Toman amount', async () => {
    const { resolver, paymentService, orderService } = setupResolver();

    // Server total = $100 (COD) + $54.50 (Online) = $154.50 * 60,000 = 9,270,000 Tomans
    const expectedTomans = Math.round(154.5 * 60000);

    const res = await resolver.requestZarinPalPayment({
      amount: expectedTomans,
      callback_url: 'https://saintbala.com/payment-callback',
      description: 'Batch Order Payment',
      currency: 'IRT',
      orderIds: ['ord-cod', 'ord-online'],
    });

    expect(orderService.recalculateBatchOrdersShipping).toHaveBeenCalledTimes(1);
    expect(paymentService.requestPayment).toHaveBeenCalledWith(
      expect.objectContaining({
        amount: 9270000,
      }),
    );
    expect(res.authority).toBeDefined();
  });

  it('rejects payment request when client amount omits PAY_ONLINE shipping cost', async () => {
    const { resolver } = setupResolver();

    // Client tries to pay only item subtotals ($150 * 60,000 = 9,000,000) without the $4.50 online shipping
    const tamperedClientAmount = Math.round(150 * 60000);

    await expect(
      resolver.requestZarinPalPayment({
        amount: tamperedClientAmount,
        callback_url: 'https://saintbala.com/payment-callback',
        description: 'Tampered Payment',
        currency: 'IRT',
        orderIds: ['ord-cod', 'ord-online'],
      }),
    ).rejects.toThrow(BadRequestException);
  });
});
